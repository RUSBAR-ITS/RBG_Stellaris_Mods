# The event text calls personal fields on the selected leader. The returned
# localization can then read the same leader's properties directly.
defined_text = {
	name = rb_le_gender
	random = no
	default = indeterminable
	text = {
		trigger = { gender = male }
		localization_key = male
	}
	text = {
		trigger = { gender = female }
		localization_key = female
	}
}

defined_text = {
	name = rb_le_species_line
	random = no
	default = BLANK_STRING
	text = {
		trigger = { exists = species }
		localization_key = rb_le_dossier_species
	}
}

# Species scope. Some unique species are neither robotic nor biological/lithoid.
defined_text = {
	name = rb_le_trait_type
	random = no
	default = rb_le_nonrobotic_type
	text = {
		trigger = { is_robotic = yes }
		localization_key = SYNTH
	}
	text = {
		trigger = { OR = { is_archetype = BIOLOGICAL is_archetype = LITHOID } }
		localization_key = rb_le_organic_type
	}
}

defined_text = {
	name = rb_le_ethic_line
	random = no
	default = BLANK_STRING
	text = {
		trigger = {
			OR = {
				is_authoritarian = yes
				is_egalitarian = yes
				is_materialist = yes
				is_spiritualist = yes
				is_militarist = yes
				is_pacifist = yes
				is_xenophile = yes
				is_xenophobe = yes
				has_ethic = ethic_gestalt_consciousness
			}
		}
		localization_key = rb_le_dossier_ethic
	}
}

# Called from the event so the saved mobile colony is available by name.
defined_text = {
	name = rb_le_mobile_home_line
	random = no
	default = BLANK_STRING
	text = {
		trigger = {
			event_target:rb_le_selected_leader = { NOT = { exists = background_planet } exists = background_colony }
			exists = event_target:rb_le_home_colony
		}
		localization_key = rb_le_dossier_home_colony
	}
}
