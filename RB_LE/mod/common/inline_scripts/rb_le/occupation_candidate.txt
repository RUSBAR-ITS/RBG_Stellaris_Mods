	limit = {
		event_target:rb_le_selected_leader = { NOT = { has_background_job = $JOB$ } }
		rb_le_job_$JOB$ = yes
	}
	change_variable = { which = rb_le_job_count value = 1 }
	change_variable = { which = rb_le_job_cursor value = 1 }

		limit = {
			check_variable = { which = rb_le_job_cursor value >= 1 }

	}
