# Country scope. Preview variables are reused when switching direction.
rb_le_prepare_age_choices = {
	set_variable = { which = rb_le_age_after_1 value = event_target:rb_le_selected_leader.value:rb_le_age_preview|DELTA|$SIGN$1| }
	set_variable = { which = rb_le_age_after_5 value = event_target:rb_le_selected_leader.value:rb_le_age_preview|DELTA|$SIGN$5| }
	set_variable = { which = rb_le_age_after_10 value = event_target:rb_le_selected_leader.value:rb_le_age_preview|DELTA|$SIGN$10| }
	set_variable = { which = rb_le_age_after_50 value = event_target:rb_le_selected_leader.value:rb_le_age_preview|DELTA|$SIGN$50| }
	set_variable = { which = rb_le_age_after_100 value = event_target:rb_le_selected_leader.value:rb_le_age_preview|DELTA|$SIGN$100| }
}

rb_le_return_to_register = {
	if = {
		limit = { check_variable = { which = rb_le_register_id value = 2 } }
		country_event = { id = rb_le.2 }
	}
	else_if = {
		limit = { check_variable = { which = rb_le_register_id value = 3 } }
		country_event = { id = rb_le.3 }
	}
	else_if = {
		limit = { check_variable = { which = rb_le_register_id value = 4 } }
		country_event = { id = rb_le.4 }
	}
	else = { country_event = { id = rb_le.5 } }
}
