pce_select_feature = {
	if = {
		limit = { has_modifier = resort_colony }
		if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_resort_spectrum_galleries }
		}
		else = {
			pce_use_feature = { TYPE = pce_resort_terraces }
		}
	}
	else_if = {
		limit = { has_modifier = penal_colony }
		if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_penal_subterranean_cell_blocks }
		}
		else = {
			pce_use_feature = { TYPE = pce_penal_tiered_prison_yards }
		}
	}
	else_if = {
		limit = { has_modifier = slave_colony }
		if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_thrall_numbered_barracks }
		}
		else = {
			pce_use_feature = { TYPE = pce_thrall_barrack_tiers }
		}
	}
	else_if = {
		limit = { uses_district_set = nomad }
		if = {
			limit = { owner = { pce_active_order = yes } has_zone = { zone = zone_ark_knights } }
			pce_use_feature = { TYPE = pce_arkship_galleries_of_the_order }
		}
		else_if = {
			limit = { owner = { has_origin = origin_sacred_path } has_zone = { zone = zone_ark_sacred_path_pilgrims } }
			pce_use_feature = { TYPE = pce_arkship_pilgrims_courts }
		}
		else_if = {
			limit = { owner = { is_hospitality_servitor_empire = yes } }
			pce_use_feature = { TYPE = pce_arkship_cabins_with_a_private_dawn }
		}
		else_if = {
			limit = { owner = { has_origin = origin_forever_cruise NOT = { has_country_flag = forever_cruise_over } } }
			pce_use_feature = { TYPE = pce_arkship_promenade_decks }
		}
		else_if = {
			limit = { owner = { has_valid_civic = civic_machine_servitor } }
			pce_use_feature = { TYPE = pce_arkship_family_decks }
		}
		else_if = {
			limit = { owner = { is_hive_empire = yes } }
			pce_use_feature = { TYPE = pce_arkship_brood_nests_under_way }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			pce_use_feature = { TYPE = pce_arkship_rotary_racks }
		}
		else_if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_arkship_custom_built_cabins }
		}
		else = {
			pce_use_feature = { TYPE = pce_arkship_streets_between_decks }
		}
	}
	else_if = {
		limit = { uses_district_set = cosmogenesis_world }
		pce_use_feature = { TYPE = pce_synaptic_lathe_pod_galleries }
	}
	else_if = {
		limit = { is_planet_class = pc_nanotech }
		pce_use_feature = { TYPE = pce_nanite_tiers }
	}
	else_if = {
		limit = { uses_district_set = habitat }
		if = {
			limit = { owner = { pce_active_order = yes } has_zone = { zone = zone_habitat_knights } }
			pce_use_feature = { TYPE = pce_habitat_galleries_of_the_order }
		}
		else_if = {
			limit = { owner = { has_valid_civic = civic_machine_servitor } }
			pce_use_feature = { TYPE = pce_habitat_starlight_cabins }
		}
		else_if = {
			limit = { owner = { is_hive_empire = yes } }
			pce_use_feature = { TYPE = pce_habitat_brood_compartments }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			pce_use_feature = { TYPE = pce_habitat_rack_mounted_decks }
		}
		else_if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_habitat_airlock_workshops }
		}
		else = {
			pce_use_feature = { TYPE = pce_habitat_modular_quarters }
		}
	}
	else_if = {
		limit = { uses_district_set = ring_world }
		if = {
			limit = { owner = { has_valid_civic = civic_machine_servitor } }
			pce_use_feature = { TYPE = pce_ringworld_garden_quays }
		}
		else_if = {
			limit = { owner = { is_hive_empire = yes } }
			pce_use_feature = { TYPE = pce_ringworld_nests_in_the_ring_frame }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			pce_use_feature = { TYPE = pce_ringworld_maintenance_tiers }
		}
		else_if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_ringworld_settlements_on_the_outer_rim }
		}
		else = {
			pce_use_feature = { TYPE = pce_ringworld_cities_above_the_main_line }
		}
	}
	else_if = {
		limit = {
			uses_district_set = hive_world
			OR = {
				owner = { is_hive_empire = yes }
				AND = {
					owner = { is_gestalt = no }
					any_owned_pop_group = { has_trait = trait_hive_mind }
				}
			}
		}
		if = {
			limit = { owner = { is_hive_empire = yes } }
			pce_use_feature = { TYPE = pce_hive_world_deep_hive_cells }
		}
		else = {
			pce_use_feature = { TYPE = pce_hive_world_reclaimed_galleries }
		}
	}
	else_if = {
		limit = { uses_district_set = machine_world }
		if = {
			limit = { owner = { has_valid_civic = civic_machine_servitor } }
			pce_use_feature = { TYPE = pce_machine_world_quiet_decks }
		}
		else_if = {
			limit = { owner = { is_hive_empire = yes } }
			pce_use_feature = { TYPE = pce_machine_world_heat_exchanger_nests }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			pce_use_feature = { TYPE = pce_machine_world_layers_of_the_planetary_machine }
		}
		else_if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_machine_world_artisans_quarters }
		}
		else = {
			pce_use_feature = { TYPE = pce_machine_world_habitation_sections }
		}
	}
	else_if = {
		limit = { uses_district_set = city_world }
		if = {
			limit = { owner = { has_valid_civic = civic_machine_servitor } }
			pce_use_feature = { TYPE = pce_ecumenopolis_garden_arcologies }
		}
		else_if = {
			limit = { owner = { is_hive_empire = yes } }
			pce_use_feature = { TYPE = pce_ecumenopolis_nest_towers }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			pce_use_feature = { TYPE = pce_ecumenopolis_assembly_towers }
		}
		else_if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_ecumenopolis_modular_arcologies }
		}
		else = {
			pce_use_feature = { TYPE = pce_ecumenopolis_upper_city }
		}
	}
	else_if = {
		limit = { uses_district_set = volcanic_world }
		if = {
			limit = { owner = { is_hive_empire = yes } }
			pce_use_feature = { TYPE = pce_volcanic_tiered_nests }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			pce_use_feature = { TYPE = pce_volcanic_distributed_nodes }
		}
		else_if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_volcanic_modular_quarters }
		}
		else = {
			pce_use_feature = { TYPE = pce_volcanic_multilayered_cities }
		}
	}
	else = {
		if = {
			limit = { owner = { is_wilderness_empire = yes } }
			pce_use_feature = { TYPE = pce_biosphere_layers }
		}
		else_if = {
			limit = { owner = { is_hive_empire = yes } }
			pce_use_feature = { TYPE = pce_tiered_nests }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			pce_use_feature = { TYPE = pce_distributed_nodes }
		}
		else_if = {
			limit = { owner = { is_synthetic_empire = yes } }
			pce_use_feature = { TYPE = pce_modular_quarters }
		}
		else = {
			pce_use_feature = { TYPE = pce_multilayered_cities }
		}
	}
}
