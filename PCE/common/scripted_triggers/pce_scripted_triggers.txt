# The decision and every menu action are restricted to the initiating human owner.
pce_valid_colony = {
	exists = colony
	owner = { is_ai = no }
}

pce_valid_session = {
	exists = event_target:pce_operator
	is_owned_by = event_target:pce_operator
	pce_valid_colony = yes
}

pce_has_capacity = {
	is_variable_set = pce_extra_districts
	check_variable = { which = pce_extra_districts value > 0 }
}

pce_amount_available = {
	pce_valid_session = yes
	OR = {
		always = $INCREASE$
		AND = {
			pce_has_capacity = yes
			check_variable = { which = pce_extra_districts value >= $AMOUNT$ }
		}
	}
}

pce_active_order = {

	NOT = { has_country_flag = quest_abandoned }
