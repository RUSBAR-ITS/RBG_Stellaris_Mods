carrier_event = {
	id = pce.1
	title = pce_menu_title
	desc = pce_main_desc
	picture = GFX_evt_pce_capacity_extension
	show_sound = event_administrative_work
	is_triggered_only = yes
	trigger = { pce_valid_session = yes }
	immediate = {
		if = { limit = { pce_has_capacity = yes } pce_reconcile_features = yes }
	}

	option = {
		name = pce_open_add
		hidden_effect = {
			if = { limit = { pce_valid_session = yes } carrier_event = { id = pce.2 } }
		}
	}
	option = {
		name = pce_open_remove
		trigger = { pce_has_capacity = yes }
		hidden_effect = {
			if = { limit = { pce_valid_session = yes } carrier_event = { id = pce.3 } }
		}
	}
	option = { name = EXIT }
}

	option = {
		name = BACK
