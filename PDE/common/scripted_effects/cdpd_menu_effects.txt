		country_event = { id = $SCREEN$ }
	}
	else = {
		country_event = { id = CDPD.3 }
	}
}

cdpd_begin_count = {
	set_variable = { which = cdpd_size value = $SIZE$ }
	set_variable = { which = cdpd_count value = 0 }
	set_variable = { which = cdpd_pages value = 0 }
	set_variable = { which = cdpd_used_slots value = 0 }
}

cdpd_assign_rank = {
	change_variable = { which = cdpd_count value = 1 }
	set_variable = { which = $ID$_rank value = cdpd_count }
	# Count pages at their first entry, using only integer operations.
	if = {
		limit = { check_variable = { which = cdpd_used_slots value = 0 } }
		change_variable = { which = cdpd_pages value = 1 }
	}
	change_variable = { which = cdpd_used_slots value = 1 }
	if = {
		limit = { check_variable = { which = cdpd_used_slots value = cdpd_size } }
		set_variable = { which = cdpd_used_slots value = 0 }
	}
}

cdpd_set_bounds = {
	if = {
		limit = { check_variable = { which = cdpd_page value > cdpd_pages } }
		set_variable = { which = cdpd_page value = cdpd_pages }
	}
	set_variable = { which = cdpd_offset value = 0 }
	set_variable = { which = cdpd_last value = 0 }
	if = {
		limit = { check_variable = { which = cdpd_pages value > 0 } }
		if = {
			limit = { check_variable = { which = cdpd_page value < 1 } }
			set_variable = { which = cdpd_page value = 1 }
		}
		set_variable = { which = cdpd_offset value = cdpd_page }
		change_variable = { which = cdpd_offset value = -1 }
		multiply_variable = { which = cdpd_offset value = cdpd_size }
		set_variable = { which = cdpd_last value = cdpd_offset }
		change_variable = { which = cdpd_last value = cdpd_size }
		if = {
			limit = { check_variable = { which = cdpd_last value > cdpd_count } }
			set_variable = { which = cdpd_last value = cdpd_count }
		}
	}
}
