cdpd_decision_add_forceful_winds_effect = {
	add_deposit = d_forceful_winds
	if = {
		limit = { has_modifier = hazardous_weather }
		remove_modifier = hazardous_weather
		set_planet_flag = cdpd_removed_hazardous_weather_for_forceful_winds
	}
}

cdpd_decision_add_genesis_preserve_effect = {
	add_deposit = d_genesis_preserve
	if = {
		limit = {
			exists = owner
			NOT = {
				any_owned_species = {
					has_genesis_preserve_trait = yes
				}
			}
		}
		owner = {
			change_variable = {
				which = presapients_guided
				value = 1
			}
		}
		spawn_presapient_species_randomizer_homeworld_effect = yes
		last_created_species = {
			set_species_flag = cdpd_genesis_preserve_species@prev
		}
		every_owned_pop_group = {
			limit = {
				species = { has_species_flag = cdpd_genesis_preserve_species@prev }
				is_sapient = no
			}
			set_pop_group_flag = cdpd_genesis_preserve_pop
		}
		set_planet_flag = cdpd_added_genesis_preserve_life
		set_planet_flag = pre_sapient_planet
	}
}

cdpd_decision_add_two_aura_blockers_effect = {
	while = {
		count = 2
		add_blocker = { type = d_aura_blocker }
	}
}
