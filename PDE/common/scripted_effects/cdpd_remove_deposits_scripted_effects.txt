# CDPD 0.10.0 removal effects. One click removes one deposit instance.

cdpd_remove_linked_d_hab_alloy_1_effect = {
	solar_system = {
		every_system_planet = {
			limit = { is_planet_class = pc_habitat }
			random_deposit = {
				limit = {
					is_deposit_type = d_hab_alloy_1
					has_deposit_flag = planet@event_target:cdpd_target_planet
				}
				remove_deposit = yes
			}
		}
	}
}

cdpd_remove_d_genesis_preserve_effect = {
	random_deposit = {
		limit = { is_deposit_type = d_genesis_preserve }
		remove_deposit = yes
	}
	while = {
		count = 3
		if = {
			limit = {
				any_owned_pop_group = {
					has_pop_group_flag = cdpd_genesis_preserve_pop
					is_sapient = no
				}
			}
			random_owned_pop_group = {
				limit = {
					has_pop_group_flag = cdpd_genesis_preserve_pop
					is_sapient = no
				}
				kill_pop_group = {
					pop_group = this
					percentage = 1
				}
			}
		}
	}
	if = {
		limit = {
			has_planet_flag = cdpd_added_genesis_preserve_life
			exists = owner
		}
		owner = {
			change_variable = {
				which = presapients_guided
				value = -1
			}
		}
		remove_planet_flag = cdpd_added_genesis_preserve_life
	}
	if = {
		limit = {
			NOT = { any_owned_pop_group = { is_sapient = no } }
		}
		remove_planet_flag = pre_sapient_planet
	}
}
