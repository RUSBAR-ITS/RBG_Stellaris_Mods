cdpd_terraforming_add_radiotrophic_preserve_effect = {
	set_country_flag = found_radio_plants
	event_target:cdpd_target_planet = {
		add_deposit = d_radiotrophic_preserve
		create_species = {
			name = random
			class = PLANT
			portrait = random
			homeworld = event_target:cdpd_target_planet
			namelist = "FUN4"
			sapient = no
			traits = {
				ideal_planet_class = "pc_nuked"
				trait = "trait_plantoid_radiotrophic"
				trait = "trait_ingenious"
				trait = "trait_fleeting"
				trait = "trait_deviants"
			}
			effect = {
				save_event_target_as = cdpd_radio_plants
			}
		}
		while = {
			count = 4
			create_pop_group = {
				species = event_target:cdpd_radio_plants
			}
		}
	}
}

cdpd_terraforming_add_explosive_atmosphere_effect = {
	event_target:cdpd_target_planet = {
		add_deposit = d_volatile_motes_1
		last_added_deposit = {
			set_deposit_flag = cdpd_linked_d_explosive_atmosphere
		}
		add_deposit = d_explosive_atmosphere
	}
	cdpd_terraforming_sync_d_volatile_motes_1_habitat_feature = yes
}

cdpd_terraforming_sync_d_volatile_motes_1_habitat_feature = {
	event_target:cdpd_target_planet.solar_system = {
		every_system_planet = {
			limit = {
				is_planet_class = pc_habitat
			}
			add_deposit = d_hab_mote_1
			last_added_deposit = {
				set_deposit_flag = planet@event_target:cdpd_target_planet
			}
		}
	}
}
