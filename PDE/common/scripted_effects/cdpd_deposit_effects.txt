cdpd_increment_counter = {
	if = {
		limit = {
			is_variable_set = $COUNTER$
		}
		change_variable = { which = $COUNTER$ value = 1 }
	}
	else = {
		set_variable = { which = $COUNTER$ value = 1 }
	}
}

cdpd_decrement_counter = {
	if = {
		limit = {
			is_variable_set = $COUNTER$
		}
		if = {
			limit = {
				check_variable = { which = $COUNTER$ value > 1 }
			}
			change_variable = { which = $COUNTER$ value = -1 }
		}
		else = {
			clear_variable = $COUNTER$
		}
	}
}

	random_deposit = {
		limit = { is_deposit_type = $DEPOSIT$ }
		remove_deposit = yes
	}
}

		limit = {
			NOT = { has_deposit = d_titanic_life_blocker }
		}
		if = {
			limit = {
				has_modifier = titanic_life
			}
			set_carrier_flag = cdpd_titanic_life_independent
		}
		else = {
			remove_carrier_flag = cdpd_titanic_life_independent
		}
	}
	add_deposit = d_titanic_life_blocker
	if = {
		limit = {
			NOT = { has_modifier = titanic_life }
		}
		remove_carrier_flag = cdpd_titanic_life_independent
		add_modifier = { modifier = titanic_life days = -1 }
	}
}

		limit = {
			NOT = { has_deposit = d_titanic_life_blocker }
		}
		if = {
			limit = {
				has_modifier = titanic_life
				NOT = { has_carrier_flag = cdpd_titanic_life_independent }
			}
			remove_modifier = titanic_life
		}
		remove_carrier_flag = cdpd_titanic_life_independent
