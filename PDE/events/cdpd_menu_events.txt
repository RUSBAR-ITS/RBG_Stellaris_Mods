namespace = CDPD

country_event = {
	id = CDPD.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_ai = no
		NOT = { has_country_flag = cdpd_session_open }
		exists = event_target:cdpd_target
		event_target:cdpd_target = { cdpd_target_in_borders = { COUNTRY = root } }
	}
	immediate = {
		set_country_flag = cdpd_session_open
		set_variable = { which = cdpd_page value = 1 }
		country_event = { id = CDPD.1002 }
	}
}

country_event = {

country_event = {
	id = CDPD.1002
	is_triggered_only = yes
	hide_window = yes
	immediate = {
		if = {
			limit = {
				cdpd_live_target = yes
			}
			country_event = { id = CDPD.2 }
		}
		else = {
			country_event = { id = CDPD.3 }
		}
	}
}
