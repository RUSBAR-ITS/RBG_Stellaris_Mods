namespace = CDPD

country_event = {
	id = CDPD.1004
	is_triggered_only = yes
	hide_window = yes
	immediate = {
		if = {
			limit = {
				cdpd_live_target = yes
			}
			if = {
				limit = {
					event_target:cdpd_target = {
						cdpd_section_remove_available = yes
					}
				}
				country_event = { id = CDPD.4 }
			}
			else = {
				set_variable = { which = cdpd_page value = 1 }
				country_event = { id = CDPD.1002 }
			}
		}
		else = {
			country_event = { id = CDPD.3 }
		}
	}
}
